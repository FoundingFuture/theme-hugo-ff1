{{/* Thumbnails that open full size when clicked. {{< gallery >}} front.png | The front page depth.png | Three levels down {{< /gallery >}} One picture per line: the file first, the caption after a pipe. Files are page resources, so put them in the page's own folder beside its index.md. Hugo cuts the thumbnails. No script. The full-size view is a :target, so clicking a thumbnail changes the fragment and CSS does the rest. Back closes it, and the link to any picture can be sent to somebody. */}} {{ $id := printf "g%d" .Ordinal }} {{ $rows := slice }} {{- range (split (trim .Inner "\n") "\n") }} {{- $cut := split . "|" }} {{- $src := trim (index $cut 0) " " }} {{- $cap := "" }}{{ if ge (len $cut) 2 }}{{ $cap = trim (index $cut 1) " " }}{{ end }} {{- with $src }}{{ $rows = $rows | append (dict "src" . "cap" $cap) }}{{ end }} {{- end }} {{ $n := len $rows }}
{{- range $i, $r := $rows }} {{- $res := $.Page.Resources.GetMatch $r.src }} {{- with $res }} {{- /* Fill, not Fit: one tall picture among wide ones makes a ragged grid three rows deep. Anchored top, because the top of a screen is the part that says which screen it is. */}} {{- $thumb := .Fill "560x380 top webp q82" }} {{ $r.cap }} {{- end }} {{- end }}
{{- range $i, $r := $rows }} {{- $res := $.Page.Resources.GetMatch $r.src }} {{- with $res }} {{- $full := .Fit "1600x1600 webp q86" }} {{- $prev := printf "#%s-%d" $id (cond (eq $i 0) (sub $n 1) (sub $i 1)) }} {{- $next := printf "#%s-%d" $id (cond (eq $i (sub $n 1)) 0 (add $i 1)) }}
{{- if gt $n 1 }} ‹ › {{- end }} ×
{{- end }} {{- end }}
